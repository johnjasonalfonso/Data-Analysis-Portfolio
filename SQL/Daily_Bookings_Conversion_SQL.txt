WITH LEADS_AGGREGATED AS (
    SELECT 
        TO_DATE(EDS_CREATE_DATE, 'YYYYMMDD') AS GEN_DATE,
        DECILE, 
        FTB_TAG, 
        CASE 
            WHEN FTB_TAG = 0 AND DECILE BETWEEN 5 AND 10 THEN 'ORIGINAL'
            WHEN FTB_TAG = 1 AND DECILE IN (7, 8, 9, 10) THEN 'ORIGINAL'
            WHEN (FTB_TAG = 1 AND DECILE IN (5, 6)) AND (GEN_DATE >= '2026-04-03') THEN 'DELTA'
            ELSE 'OTHER'
        END AS WHITELIST_TAG,
        COUNT(DISTINCT CIF) AS RAW_LEADS_VOL
    FROM MP_QUICKLOANS.QL_APPROVED.NEW_CONVERSION_CIF
    WHERE PRODUCT_SOURCE = 'PAYROLL' 
    GROUP BY 1, 2, 3, 4
),

LEADS_CUMULATIVE AS (
    SELECT *,
        SUM(RAW_LEADS_VOL) OVER (
            PARTITION BY EXTRACT(YEAR FROM GEN_DATE), EXTRACT(MONTH FROM GEN_DATE), 
                         DECILE, FTB_TAG
            ORDER BY GEN_DATE
        ) AS TOTAL_LEADS_VOL,
        COALESCE(DATEADD('day', -1, LEAD(GEN_DATE) OVER (
            PARTITION BY DECILE, FTB_TAG, WHITELIST_TAG
            ORDER BY GEN_DATE
        )), '2099-12-31') AS WINDOW_END
    FROM LEADS_AGGREGATED
    WHERE WHITELIST_TAG != 'OTHER'
),

LEADS_DAILY_MASTER AS (
    SELECT 
        D.CALENDAR_DATE, 
        L.DECILE, 
        L.FTB_TAG,
        L.TOTAL_LEADS_VOL, 
        L.GEN_DATE AS SOURCE_GEN_DATE,
        L.WINDOW_END
    FROM (SELECT DISTINCT BOOKINGDATE AS CALENDAR_DATE FROM MP_QUICKLOANS.QL_APPROVED.NEW_VINTAGE_DELINQUENCY_LOAN_LEVEL) D
    INNER JOIN LEADS_CUMULATIVE L 
        ON D.CALENDAR_DATE BETWEEN L.GEN_DATE AND L.WINDOW_END
),

BOOKINGS_WITH_TERMS AS (
    SELECT
        B.BOOKINGDATE, 
        B.DECILE AS ATTR_DECILE, 
        B.FTB_TAG AS ATTR_FTB,
        C.MAX_TENOR_WL AS OFFERED_TERM,
        C.LOANTERM_BOOKED AS BOOKED_TERM,
        C.INTEREST_RATE_WL AS OFFERED_RATE, -- Added from Conversion Table
        COUNT(DISTINCT B.FINACLELOANNUMBER) AS TOTAL_BOOKINGS_SUM,
        SUM(B.LOANAMOUNT) AS TOTAL_LOAN_AMT_SUM
    FROM MP_QUICKLOANS.QL_APPROVED.NEW_VINTAGE_DELINQUENCY_LOAN_LEVEL B
    LEFT JOIN (
        -- Take latest record per loan to prevent fan-out bloat
        SELECT *, ROW_NUMBER() OVER (PARTITION BY FINACLELOANNUMBER ORDER BY EDS_CREATE_DATE DESC) as rnk
        FROM MP_QUICKLOANS.QL_APPROVED.NEW_CONVERSION_CIF
    ) C ON B.FINACLELOANNUMBER = C.FINACLELOANNUMBER AND C.rnk = 1
    WHERE B.PRODUCTGROUP = 'PAYROLL'
    GROUP BY 1, 2, 3, 4, 5, 6
)

---------------------------------------------------------
-- FINAL STEP: JOIN ON THE DAILY MASTER TIMELINE
---------------------------------------------------------

SELECT 
    COALESCE(M.CALENDAR_DATE, B.BOOKINGDATE) AS ACTIVITY_DATE,
    EXTRACT(YEAR FROM ACTIVITY_DATE) AS BK_YEAR,
    EXTRACT(MONTH FROM ACTIVITY_DATE) AS BK_MONTH,
    EXTRACT(DAY FROM ACTIVITY_DATE) AS BK_DAY,
    COALESCE(M.DECILE, B.ATTR_DECILE) AS DECILE,
    COALESCE(M.FTB_TAG, B.ATTR_FTB) AS FTB_TAG,
    
    B.OFFERED_TERM,
    B.BOOKED_TERM,
    B.OFFERED_RATE,

    CASE 
        WHEN M.SOURCE_GEN_DATE IS NULL THEN 'NON_WHITELISTED'
        WHEN B.BOOKINGDATE IS NULL THEN 'NO_BOOKING'
        WHEN B.OFFERED_TERM != B.BOOKED_TERM THEN 'TERM_VARIANCE'
        ELSE 'CONVERTED'
    END AS BEHAVIOR_TAG,


    CASE 
        WHEN ROW_NUMBER() OVER (
            PARTITION BY ACTIVITY_DATE, DECILE, FTB_TAG
            ORDER BY B.TOTAL_BOOKINGS_SUM DESC NULLS LAST
        ) = 1 THEN ZEROIFNULL(M.TOTAL_LEADS_VOL) 
        ELSE 0 
    END AS LEADS_BASELINE,

    ZEROIFNULL(B.TOTAL_BOOKINGS_SUM) AS TOTAL_BOOKINGS,
    ZEROIFNULL(B.TOTAL_LOAN_AMT_SUM) AS LOAN_AMOUNT
FROM LEADS_DAILY_MASTER M
FULL OUTER JOIN BOOKINGS_WITH_TERMS B 
    ON  M.CALENDAR_DATE = B.BOOKINGDATE
    AND M.DECILE = B.ATTR_DECILE
    AND M.FTB_TAG = B.ATTR_FTB
ORDER BY 1 DESC, 2 ASC;
